// Laço com dum: fatorial iterativo
funct int fatorial(int n) {
    int resultado = 1;
    int i = 2;
    dum (i <= n) {
        resultado = resultado * i;
        i = i + 1;
    }
    redire resultado;
}

// Parênteses e divisão; usa float
funct float media(float a, float b) {
    redire (a + b) / 2;
}

// Condicional com redire em cada caminho
funct int maior(int a, int b) {
    si (a > b) {
        redire a;
    }
    redire b;
}

// Recursão
funct int fibonacci(int n) {
    si (n <= 1) {
        redire n;
    }
    redire fibonacci(n - 1) + fibonacci(n - 2);
}
